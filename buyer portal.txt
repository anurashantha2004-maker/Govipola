        function applyFilters() {
            const cards = document.querySelectorAll('.product-card');
            let visibleCount = 0;

            cards.forEach(card => {
                const produce = card.getAttribute('data-produce');
                const region = card.getAttribute('data-region');
                const qty = parseInt(card.getAttribute('data-quantity'));

                const produceMatch = filters.produce.value === 'all' || filters.produce.value === produce;
                const regionMatch = filters.region.value === 'all' || filters.region.value === region;
                
                let qtyMatch = true;
                const qtyVal = filters.quantity.value;
                if (qtyVal === '0-100') qtyMatch = qty <= 100;
                else if (qtyVal === '100-500') qtyMatch = qty > 100 && qty <= 500;
                else if (qtyVal === '500-1000') qtyMatch = qty > 500 && qty <= 1000;
                else if (qtyVal === '1000+') qtyMatch = qty > 1000;

                if (produceMatch && regionMatch && qtyMatch) {
                    card.style.display = 'block';
                    visibleCount++;
                } else {
                    card.style.display = 'none';
                }
            });

            document.getElementById('result-count').textContent = `(${visibleCount} items)`;
